Arquitectura de la solución

Flujo completo de datos

🔄
Flujo:
1. Usuario hace clic en "Descargar reporte" en el dashboard
2. El host llama al endpoint POST /api/reports/transactions
3. El host invoca la Lambda (via HTTP / SDK)
4. La Lambda consulta la API de transacciones y genera el PDF
5. Guarda el PDF en S3 con un nombre único
6. Retorna una URL pre-firmada de S3 (válida por 1 hora)
7. El navegador descarga el PDF directamente desde S3

📦 Dependencias Lambda

pdfkit — generación de PDF
@aws-sdk/client-s3 — subir a S3
node-fetch — HTTP requests (incluido en Node 18+)

🗄️ Storage: S3

Los PDFs se guardan en S3 con nombres únicos (transactions-{timestamp}.pdf). Se generan URLs pre-firmadas para descarga segura sin exponer el bucket.

Paso 3 – Obtener las transacciones

La Lambda consulta la API del host Next.js

🔌
Módulo fetchTransactions
lib/fetchTransactions.mjs
/**
 * Obtiene las transacciones desde la API del host.
 * Parámetros opcionales para filtrar por fecha.
 */
export async function fetchTransactions({ startDate, endDate } = {}) {
  const HOST_API_URL = process.env.HOST_API_URL;

  if (!HOST_API_URL) {
    throw new Error("HOST_API_URL no está configurada");
  }

  // Construir query params opcionales
  const params = new URLSearchParams();
  if (startDate) params.set("startDate", startDate);
  if (endDate)   params.set("endDate", endDate);

  const url = `${HOST_API_URL}/api/transactions?${params}`;
  console.log("Fetching transactions from:", url);

  const response = await fetch(url, {
    headers: {
      // Cabecera interna para que la API sepa que es una llamada de Lambda
      "X-Internal-Token": process.env.INTERNAL_API_TOKEN ?? "",
      "Content-Type": "application/json",
    },
  });

  if (!response.ok) {
    throw new Error(`API error: ${response.status} ${response.statusText}`);
  }

  const transactions = await response.json();
  console.log(`Obtenidas ${transactions.length} transacciones`);
  return transactions;
}
🔐
El header X-Internal-Token es opcional pero recomendado. En tu API de Next.js puedes verificarlo para asegurarte de que la llamada viene de tu Lambda y no de un tercero.

Paso 4 – Generar el PDF con PDFKit

Construye el reporte con formato profesional

📄
Módulo generatePdf
lib/generatePdf.mjs
import PDFDocument from "pdfkit";

const formatCurrency = (amount) =>
  new Intl.NumberFormat("en-US", { style: "currency", currency: "USD" }).format(amount);

const formatDate = (iso) =>
  new Date(iso).toLocaleDateString("es-MX", { year: "numeric", month: "long", day: "numeric" });

/**
 * Genera el PDF de transacciones en memoria como Buffer.
 * @param {Array} transactions - Lista de transacciones de MongoDB
 * @returns {Promise<Buffer>}
 */
export function generateTransactionsPdf(transactions) {
  return new Promise((resolve, reject) => {
    const doc = new PDFDocument({ margin: 50, size: "A4" });
    const chunks = [];

    doc.on("data", (chunk) => chunks.push(chunk));
    doc.on("end",  () => resolve(Buffer.concat(chunks)));
    doc.on("error", reject);

    // ── Encabezado ──────────────────────────────────────
    doc.rect(0, 0, doc.page.width, 80).fill("#1C2536");

    doc
      .fillColor("#818cf8")
      .fontSize(22)
      .font("Helvetica-Bold")
      .text("Crypto Dashboard", 50, 22);

    doc
      .fillColor("#637381")
      .fontSize(11)
      .font("Helvetica")
      .text("Reporte de Transacciones", 50, 50);

    // Fecha del reporte (derecha)
    doc
      .fillColor("#9CA3AF")
      .fontSize(9)
      .text(`Generado: ${new Date().toLocaleString("es-MX")}`, 0, 35, { align: "right" });

    doc.moveDown(3);

    // ── Resumen estadístico ──────────────────────────────
    const totalAmount = transactions.reduce((s, t) => s + (t.amount || 0), 0);
    const yStats = doc.y;

    // Caja de resumen
    doc.rect(50, yStats, doc.page.width - 100, 60).fillAndStroke("#F4F6F8", "#E5E7EB");

    doc.fillColor("#111827").fontSize(10).font("Helvetica-Bold")
       .text("Total transacciones", 70, yStats + 12);
    doc.fillColor("#4f46e5").fontSize(18)
       .text(String(transactions.length), 70, yStats + 26);

    doc.fillColor("#111827").fontSize(10).font("Helvetica-Bold")
       .text("Volumen total", 220, yStats + 12);
    doc.fillColor("#059669").fontSize(18)
       .text(formatCurrency(totalAmount), 220, yStats + 26);

    doc.moveDown(4.5);

    // ── Encabezado de tabla ──────────────────────────────
    const yTable = doc.y;
    doc.rect(50, yTable, doc.page.width - 100, 22).fill("#1C2536");

    doc.fillColor("#9CA3AF").fontSize(8).font("Helvetica-Bold");
    [
      [60,  "#"],
      [80,  "USUARIO"],
      [220, "MONEDA"],
      [320, "MONTO"],
      [410, "FECHA"],
    ].forEach(([x, label]) => doc.text(label, x, yTable + 7));

    doc.moveDown(1.5);

    // ── Filas de transacciones ───────────────────────────
    transactions.forEach((tx, i) => {
      if (doc.y > 750) doc.addPage();  // nueva página si no hay espacio

      const yRow = doc.y;
      const isEven = i % 2 === 0;
      if (isEven) {
        doc.rect(50, yRow - 2, doc.page.width - 100, 18).fill("#FAFAFA");
      }

      doc.fillColor("#6B7280").fontSize(8).font("Helvetica")
         .text(String(i + 1), 60, yRow);

      doc.fillColor("#111827").fontSize(8)
         .text(tx.user || "—", 80, yRow, { width: 130, ellipsis: true });

      doc.fillColor("#4f46e5").fontSize(8).font("Helvetica-Bold")
         .text(tx.coin || "—", 220, yRow);

      doc.fillColor("#059669").fontSize(8)
         .text(formatCurrency(tx.amount || 0), 320, yRow);

      doc.fillColor("#6B7280").fontSize(8).font("Helvetica")
         .text(formatDate(tx.createdAt), 410, yRow);

      doc.moveDown(1.2);
    });

    // ── Footer ───────────────────────────────────────────
    doc.rect(50, doc.y + 10, doc.page.width - 100, 1).fill("#E5E7EB");
    doc.fillColor("#9CA3AF").fontSize(8)
       .text("Crypto Dashboard — Reporte generado automáticamente", 50, doc.y + 16, { align: "center" });

    doc.end();
  });
}

Paso 5 – Guardar el PDF en S3

Upload y URL pre-firmada de descarga

☁️
Módulo uploadToS3
lib/uploadToS3.mjs
import { S3Client, PutObjectCommand, GetObjectCommand } from "@aws-sdk/client-s3";
import { getSignedUrl } from "@aws-sdk/s3-request-presigner";

const s3 = new S3Client({ region: process.env.AWS_REGION ?? "us-east-1" });

/**
 * Sube el buffer del PDF a S3 y retorna una URL pre-firmada (1 hora).
 * @param {Buffer} pdfBuffer
 * @param {string} fileName - Ej: "transactions-2025-07-10.pdf"
 * @returns {Promise<string>} URL de descarga
 */
export async function uploadPdfToS3(pdfBuffer, fileName) {
  const bucket = process.env.S3_BUCKET_NAME;
  const key = `reports/${fileName}`;  // carpeta "reports/" en el bucket

  // 1. Subir el PDF
  await s3.send(new PutObjectCommand({
    Bucket: bucket,
    Key: key,
    Body: pdfBuffer,
    ContentType: "application/pdf",
    ContentDisposition: `attachment; filename="${fileName}"`,
    // Metadata útil para buscar en S3
    Metadata: {
      "generated-at": new Date().toISOString(),
      "generated-by": "lambda-pdf-report",
    },
  }));

  console.log(`PDF subido a s3://${bucket}/${key}`);

  // 2. Generar URL pre-firmada válida por 1 hora
  const signedUrl = await getSignedUrl(
    s3,
    new GetObjectCommand({ Bucket: bucket, Key: key }),
    { expiresIn: 3600 }  // 3600 segundos = 1 hora
  );

  return signedUrl;
}
🔗
La URL pre-firmada incluye credenciales temporales en los parámetros de la URL. El usuario puede descargar el PDF directamente desde S3 sin que el bucket sea público. Expira automáticamente.

Paso 6 – Handler completo

El punto de entrada que integra todo

⚡
Handler principal
index.mjs
import { fetchTransactions } from "./lib/fetchTransactions.mjs";
import { generateTransactionsPdf } from "./lib/generatePdf.mjs";
import { uploadPdfToS3 } from "./lib/uploadToS3.mjs";

export const handler = async (event) => {
  console.log("Iniciando generación de reporte de transacciones");

  try {
    // Parámetros opcionales de filtro desde el body HTTP
    const body = event.body ? JSON.parse(event.body) : {};
    const { startDate, endDate } = body;

    // 1. Obtener transacciones
    const transactions = await fetchTransactions({ startDate, endDate });

    if (transactions.length === 0) {
      return {
        statusCode: 200,
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ message: "No hay transacciones para el período" }),
      };
    }

    // 2. Generar el PDF en memoria
    console.log(`Generando PDF para ${transactions.length} transacciones...`);
    const pdfBuffer = await generateTransactionsPdf(transactions);

    // 3. Nombre único del archivo
    const timestamp = new Date().toISOString().slice(0, 10); // "2025-07-10"
    const fileName = `transactions-${timestamp}-${Date.now()}.pdf`;

    // 4. Subir a S3 y obtener URL
    console.log("Subiendo PDF a S3...");
    const downloadUrl = await uploadPdfToS3(pdfBuffer, fileName);

    console.log("Reporte generado exitosamente", { fileName, downloadUrl });

    return {
      statusCode: 200,
      headers: { "Content-Type": "application/json",
                  "Access-Control-Allow-Origin": "*" },
      body: JSON.stringify({
        success: true,
        fileName,
        downloadUrl,     // URL pre-firmada de S3
        totalTransactions: transactions.length,
        generatedAt: new Date().toISOString(),
      }),
    };

  } catch (error) {
    console.error("Error generando reporte:", error);
    return {
      statusCode: 500,
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ error: error.message }),
    };
  }
};

mkdir -p layer/nodejs
cp package.json layer/nodejs/
cd layer/nodejs && npm install --production && cd ../..
zip -r layer.zip layer/

# 2. Subir el layer a Lambda
aws lambda publish-layer-version \
  --layer-name pdf-report-dependencies \
  --zip-file fileb://layer.zip \
  --compatible-runtimes nodejs20.x \
  --region us-east-1

# 3. Empaquetar solo el código (sin node_modules)
zip -r function.zip index.mjs lib/

# 4. Subir la función
aws lambda update-function-code \
  --function-name generate-pdf-report \
  --zip-file fileb://function.zip \
  --region us-east-1

# 5. Configurar variables de entorno
aws lambda update-function-configuration \
  --function-name generate-pdf-report \
  --environment Variables="{
    HOST_API_URL=https://tu-dominio.com,
    S3_BUCKET_NAME=dashboard-reportes-pdf,
    AWS_REGION=us-east-1,
    INTERNAL_API_TOKEN=tu-token-secreto
  }" \
  --timeout 30 \
  --memory-size 512 \
  --region us-east-1

Paso 8 – Endpoint en el Host (Next.js)

Botón "Descargar Reporte" en el dashboard

🖥️
API Route en Next.js
app/api/reports/transactions/route.ts
import { NextRequest, NextResponse } from "next/server";

const LAMBDA_URL = process.env.LAMBDA_PDF_URL;
// Ej: https://xxxx.execute-api.us-east-1.amazonaws.com/generate-pdf

export async function POST(req: NextRequest) {
  try {
    const body = await req.json().catch(() => ({}));

    // Llama a la Lambda via API Gateway
    const response = await fetch(LAMBDA_URL!, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(body),
    });

    const data = await response.json();

    if (!response.ok) {
      return NextResponse.json({ error: data.error }, { status: 500 });
    }

    // Retorna la URL de descarga al cliente
    return NextResponse.json({
      downloadUrl: data.downloadUrl,
      totalTransactions: data.totalTransactions,
      fileName: data.fileName,
    });

  } catch (error) {
    return NextResponse.json(
      { error: "Error al generar el reporte" },
      { status: 500 }
    );
  }
}

Botón de descarga en la página de Transacciones (MFE)
"use client";
import { useState } from "react";

const HOST_URL = process.env.NEXT_PUBLIC_HOST_URL ?? "http://localhost:3000";

export function DownloadReportButton() {
  const [loading, setLoading] = useState(false);

  const handleDownload = async () => {
    setLoading(true);
    try {
      const res = await fetch(`${HOST_URL}/api/reports/transactions`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({}),
      });
      const data = await res.json();

      if (data.downloadUrl) {
        // Abre la URL pre-firmada de S3 directamente
        window.open(data.downloadUrl, "_blank");
      }
    } catch (e) {
      alert("Error al generar reporte");
    } finally {
      setLoading(false);
    }
  };

  return (
    <button onClick={handleDownload} disabled={loading}
      style={{ background: "#4f46e5", color: "#fff", padding: "8px 16px",
               borderRadius: "8px", border: "none", cursor: loading ? "wait" : "pointer" }}>
      {loading ? "Generando PDF…" : "⬇ Descargar Reporte"}
    </button>
  );
}
